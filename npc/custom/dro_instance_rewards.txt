//===== DimensionsRO ==========================================
//= Instance Rewards — Booster Coin per completion
//===== Description: ==========================================
//= Funcao F_GrantBoosterCoin chamada do reward block de cada
//= instance script no fim do flow (no NPC que ja entrega os
//= rewards finais — getitem/getexp).
//=
//= Tiers (arg 0):
//=   1 = Easy   (2 coins)  — Lv 50-119, clear <15min
//=   2 = Medium (5 coins)  — Lv 120-159, clear 15-30min
//=   3 = Hard   (10 coins) — Lv 160+, end-game
//=   4 = Mega   (20 coins) — Top end (OGH Hard, Mansion)
//=
//= Cap diario por conta (#-scope): 30 coins.
//= Storage: #DRO_BC_DAILY_DATE$ ("YYYY-DDD") + #DRO_BC_DAILY_AMOUNT.
//= Cap reseta no primeiro grant do novo dia.
//=
//= Quando o cap eh atingido, mostra dispbottom mas nao falha o
//= instance reward (player ainda recebe getitem/getexp normais).
//============================================================

function	script	F_GrantBoosterCoin	{
	.@tier = getarg(0, 0);
	if (.@tier < 1 || .@tier > 4) return;

	setarray .@amounts[1], 2, 5, 10, 20;
	.@reward = .@amounts[.@tier];

	.@cap = 30;
	.@today$ = sprintf("%d-%03d", gettime(7), gettime(8));
	if (#DRO_BC_DAILY_DATE$ != .@today$) {
		#DRO_BC_DAILY_DATE$ = .@today$;
		#DRO_BC_DAILY_AMOUNT = 0;
	}

	.@available = .@cap - #DRO_BC_DAILY_AMOUNT;
	if (.@available <= 0) {
		dispbottom "[Booster] Cap diario de Booster Coin atingido (" + .@cap + "/dia). Volte amanha.";
		return;
	}

	.@grant = (.@reward < .@available) ? .@reward : .@available;
	getitem Booster_Coin, .@grant;
	#DRO_BC_DAILY_AMOUNT += .@grant;
	dispbottom "[Booster] +" + .@grant + " Booster Coin (" + #DRO_BC_DAILY_AMOUNT + "/" + .@cap + " hoje).";
	return;
}
