//===== DimensionsRO ==========================================
//= Mercenary Trainer (Phase 6 -- ai-server hired support shells)
//===== Description: =========================================
//= NPC para contratar um shell-mercenario de suporte (linha
//= Acolyte/Priest/High Priest/Arch Bishop). O shell e' um BL_PC
//= controlado pelo ai-server: segue o player, cura, buffa,
//= cura status, faz Lex Aeterna em alvos do owner.
//=
//= Regras:
//=  1) So' o LIDER da party pode contratar.
//=  2) O nivel do merc e' derivado do nivel do lider (nao ha'
//=     escolha de tier — o player so escolhe a classe).
//=  3) Preco escala com o nivel do merc:
//=        zeny/min = merc_lvl * 100
//=     Ex: AB lvl 130 por 60min = 130 * 100 * 60 = 780,000z.
//=  4) Apos confirmar: cobra zeny, roda @hire, merc spawna em
//=     1-2s, entra na party automaticamente, segue/cura/buffa.
//============================================================

prt_in,44,113,4	script	Mercenary Trainer#dro	1_F_PRIEST,{
	mes "[Mercenary Trainer]";
	mes "Boa caca, aventureiro.";
	mes "Posso te emprestar um irmao da fe pra te ajudar nos campos.";
	mes " ";
	mes "Ele vai te seguir, te curar e te abencoar. Nao ataca por conta propria.";
	next;

	// Party requirement — o merc entra na party do owner; sem party nao ha
	// onde encaixa-lo.
	if (getcharid(1) == 0) {
		mes "[Mercenary Trainer]";
		mes "Voce precisa estar em uma ^FF0000party^000000 para contratar um irmao da fe.";
		mes " ";
		mes "Crie uma party (^FF0000/organize <nome>^000000) ou junte-se a uma e volte aqui.";
		close;
	}

	// Apenas o lider da party pode contratar.
	if (!is_party_leader()) {
		mes "[Mercenary Trainer]";
		mes "So o ^FF0000lider da party^000000 pode contratar um irmao da fe.";
		mes " ";
		mes "Peca para o lider conversar comigo.";
		close;
	}

	.@blvl = BaseLevel;

	// Classe automatica pelo nivel do lider; o nivel do merc casa com o
	// nivel do lider (ai-server escala via base_level_override).
	//   <  40  -> Acolyte
	//   40-69  -> Priest
	//   70-98  -> High Priest
	//   >= 99  -> Arch Bishop
	if (.@blvl < 40) {
		.@job = 4;     .@tier = 0;
		.@cls$ = "Acolyte";
		.@jlvl = 50;	// Acolyte job max
	} else if (.@blvl < 70) {
		.@job = 8;     .@tier = 1;
		.@cls$ = "Priest";
		.@jlvl = 50;	// Priest job max
	} else if (.@blvl < 99) {
		.@job = 4014;  .@tier = 1;
		.@cls$ = "High Priest";
		.@jlvl = 70;	// HP job max
	} else {
		.@job = 4063;  .@tier = 2;
		.@cls$ = "Arch Bishop";
		.@jlvl = 70;	// AB job max
	}
	.@merc_lvl = .@blvl;	// merc nasce no mesmo nivel do lider

	mes "[Mercenary Trainer]";
	mes "Pra um lider de nivel ^0000FF" + .@blvl + "^000000 vou enviar um";
	mes "^FF0000" + .@cls$ + "^000000 (lvl " + .@merc_lvl + ").";
	mes " ";
	mes "Por quanto tempo?";
	mes "(custo = nivel x 100z por minuto)";
	next;

	.@cost_5  = .@merc_lvl * 100 * 5;
	.@cost_30 = .@merc_lvl * 100 * 30;
	.@cost_60 = .@merc_lvl * 100 * 60;

	.@dur = select(
		"5 minutos (" + .@cost_5  + "z)",
		"30 minutos (" + .@cost_30 + "z)",
		"60 minutos (" + .@cost_60 + "z)",
		"Cancelar"
	);

	if (.@dur == 4) close;

	setarray .@minutes[0], 5, 30, 60;
	.@min   = .@minutes[.@dur - 1];
	.@price = .@merc_lvl * 100 * .@min;

	if (Zeny < .@price) {
		mes "[Mercenary Trainer]";
		mes "Zeny insuficiente. Custa ^0000FF" + .@price + "z^000000.";
		close;
	}

	mes "[Mercenary Trainer]";
	mes "Confirmando: irmao ^0000FF" + .@merc_lvl + "^000000 por " + .@min + " minutos. Custo: ^0000FF" + .@price + "z^000000.";
	next;
	if (select("Sim:Nao") == 2) close;

	Zeny -= .@price;
	atcommand "@hire " + .@job + " " + .@tier + " " + .@min + " " + .@merc_lvl + " " + .@jlvl;
	mes "[Mercenary Trainer]";
	mes "Que a luz te guie. Use ^FF0000@dismiss^000000 pra dispensa-lo a qualquer momento.";
	close;
}
