//===== Description: =========================================
//= Welcome NPC. Entrega 1x Booster Call Package (101538) por char.
//= Flag: DRO_STARTER_TAKEN.
//============================================================

prt_in,37,95,4	script	Mega Booster Support#dro	4_M_ALCHE_A,{
	mes "[Mega Booster Support]";
	mes "Welcome to DimensionsRO!";

	if (DRO_STARTER_TAKEN) {
		mes "Voce ja pegou seu kit inicial.";
		close;
	}

	if (checkweight(501, 1) == 0 || (MaxWeight - Weight) < 1000) {
		mes "Inventario muito cheio. Volta mais leve.";
		close;
	}

	next;
	if (select("Pegar Booster box:Cancelar") != 1) close;

	if (vip_status(VIP_STATUS_ACTIVE))
		getitem 103048, 1;
	else
		getitem 101538, 1;
	set DRO_STARTER_TAKEN, 1;

	mes "[Mega Booster Support]";
	mes "Pronto! Boa cacada.";
	close;
}
