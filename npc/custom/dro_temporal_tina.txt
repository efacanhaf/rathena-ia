//===== DimensionsRO ==========================================
//= Temporal Tina — vende Episode Pass Tickets por Nyan.
//===== Description: ==========================================
//= NPC oficial iRO ("Temporal Tina") que vende os Episode
//= Pass Tickets em troca de Nyangvine. Em DimensionsRO usamos
//= rates ajustados (kRO oficial cobra 25 Nyan/ticket — DRO
//= cobra metade ou menos, calibrado pra economia local).
//=
//= Tickets vendidos:
//=   Ep 14 Pass Ticket (1001755) -> 8 Nyan
//=   Ep 15 Pass Ticket (1001756) -> 8 Nyan
//=   Ep 16 Pass Ticket (1001260) -> 10 Nyan
//=   Ep 17 Pass Ticket (1001261) -> 12 Nyan
//=
//= Storage Nyan: item 6909 (Nyangvine_Fruit).
//=
//= Pareada com a Valkyrie (prt_cas 373,77 — quem aplica o
//= skip) e com o Cambista (prontera 158,196 — quem fornece
//= o Nyan via Cash/BC).
//=
//= Localizacao: prontera 156,196 (ao lado do Cambista).
//============================================================

prontera,156,196,5	script	Temporal Tina#dro	4_F_TELEPORTER,{
	mes "[Temporal Tina]";
	mes "Ola! Eu vendo ^4d4dffPasses de Episodio^000000.";
	mes "Cada passe permite pular a campanha principal de um episodio inteiro — voce entrega o passe pra Valkyrie no Castelo de Prontera.";
	next;
	mes "[Temporal Tina]";
	mes "Aceito ^4d4dffNyangvine^000000 como pagamento. Voce tem ^4d4dff" + countitem(6909) + " Nyangvine(s)^000000 no momento.";
	next;

	switch (select(
		"Passe Ep 14 (Lighthalzen, Bifrost, Eclage, Dewata, Malangdo, Malaya) — 8 Nyan:" +
		"Passe Ep 15 (Phantasmagorika, Reload) — 8 Nyan:" +
		"Passe Ep 16 (Banquete dos Herois, Charleston) — 10 Nyan:" +
		"Passe Ep 17 (Terra Gloria, Mansion) — 12 Nyan:" +
		"Sair")) {
	case 1: callsub L_buy, 1001755, 8, "14"; break;
	case 2: callsub L_buy, 1001756, 8, "15"; break;
	case 3: callsub L_buy, 1001260, 10, "16"; break;
	case 4: callsub L_buy, 1001261, 12, "17"; break;
	case 5: close;
	}
	close;

L_buy:
	.@item = getarg(0);
	.@cost = getarg(1);
	.@ep$ = getarg(2);

	if (countitem(6909) < .@cost) {
		mes "[Temporal Tina]";
		mes "Voce nao tem Nyangvines suficientes. Custo: ^4d4dff" + .@cost + " Nyan^000000. Voce tem: ^4d4dff" + countitem(6909) + "^000000.";
		mes "Visite o Cambista para trocar Cash ou Booster Coin por Nyangvine.";
		return;
	}
	if (checkweight(.@item, 1) == 0) {
		mes "[Temporal Tina]";
		mes "Voce nao tem espaco no inventario.";
		return;
	}
	mes "[Temporal Tina]";
	mes "Voce vai comprar ^4d4dff1 Passe de Episodio " + .@ep$ + "^000000 por ^4d4dff" + .@cost + " Nyangvine(s)^000000.";
	mes "Confirma?";
	next;
	if (select("Sim, comprar:Cancelar") == 2) {
		mes "[Temporal Tina]";
		mes "Como desejar.";
		return;
	}
	delitem 6909, .@cost;
	getitem .@item, 1;
	mes "[Temporal Tina]";
	mes "Pronto! Aproveite. Use a Valkyrie no Castelo de Prontera (prt_cas 373,77) para aplicar o skip.";
	return;
}
