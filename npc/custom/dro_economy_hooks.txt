//===== DimensionsRO ==========================================
//= Economy Hooks — daily Cash bonus.
//===== Description: ==========================================
//= 1. OnPCLoginEvent: +2 Cash no primeiro login do dia.
//=    Storage: #DRO_DAILY_CASH_LAST$ (key "AAAA-DDD" do dia do ano).
//=    Account-scoped — soh um char por conta ganha por dia, simulado
//=    pelo fato de #CASHPOINTS ser conta-wide.
//=
//= O contador de MVP -> Nyangvine (100 kills = 1 Nyan) ESTAVA aqui
//= como OnNPCKillEvent, mas migrou pra um hook C++ direto em
//= mob_dead (src/map/mob.cpp). Motivo: a maioria das linhas de
//= boss_monster do rAthena (.../re/mobs/dungeons/*.txt) terminam em
//= ",1" — o parser interpreta esse "1" como label de evento, e o
//= mob fica com npc_event="1". A doc do rAthena confirma:
//= "OnNPCKillEvent triggers when a player kills a monster *without
//= label*", entao o global event NUNCA disparava pros MVPs reais.
//= O hook C++ checa md->get_bosstype() == BOSSTYPE_MVP em mob_dead
//= antes do dispatch de label, dispensando o script trigger.
//=
//= Account reg #DRO_MVP_KILLS continua o mesmo nome — qualquer
//= valor existente persiste; so a fonte do incremento mudou.
//============================================================

-	script	dro_economy_hooks	-1,{
	end;

OnPCLoginEvent:
	// Daily Cash bonus — 2 Cash no primeiro login do dia.
	.@today$ = sprintf("%d-%03d", gettime(7), gettime(8));
	if (#DRO_DAILY_CASH_LAST$ != .@today$) {
		#CASHPOINTS += 2;
		#DRO_DAILY_CASH_LAST$ = .@today$;
		dispbottom "[DimensionsRO] +2 Cash de bonus diario. Saldo: " + (#CASHPOINTS) + " Cash.";
	}
	end;
}
